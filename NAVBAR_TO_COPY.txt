<script src="https://cdnjs.cloudflare.com/ajax/libs/gsap/3.12.2/gsap.min.js"></script>
<script>
    document.addEventListener('DOMContentLoaded', function () {
        const toggleBtn = document.getElementById("toggle-btn");
        const hamburger = document.getElementById("hamburger");
        const menu = document.querySelector(".menu");
        const menuItems = document.querySelectorAll(".menu-item a");
        const servicesToggle = document.querySelector(".services-toggle");
        const hasSubmenu = document.querySelector(".has-submenu");

        // Menu toggle functionality
        toggleBtn.onclick = (e) => {
            hamburger.classList.toggle("active");
            
            if (hamburger.classList.contains("active")) {
                // Open menu
                menu.style.visibility = "visible";
                gsap.to(menuItems, {
                    top: 0,
                    duration: 0.8,
                    stagger: 0.1,
                    ease: "power3.out"
                });
                
                setTimeout(() => {
                    hamburger.classList.add('menu-visible');
                }, 800);
            } else {
                // Close menu
                hamburger.classList.remove('menu-visible');
                gsap.to(menuItems, {
                    top: "100vh",
                    duration: 0.6,
                    stagger: 0.05,
                    ease: "power3.in",
                    onComplete: () => {
                        menu.style.visibility = "hidden";
                    }
                });
            }
        };

        // Services submenu toggle
        if (servicesToggle && hasSubmenu) {
            servicesToggle.addEventListener('click', function(e) {
                e.preventDefault();
                hasSubmenu.classList.toggle('active');
            });
        }

        // Close menu when clicking on a menu item
        menuItems.forEach(item => {
            item.addEventListener('click', function() {
                if (!this.classList.contains('services-toggle')) {
                    hamburger.classList.remove("active");
                    hamburger.classList.remove('menu-visible');
                    gsap.to(menuItems, {
                        top: "100vh",
                        duration: 0.6,
                        stagger: 0.05,
                        ease: "power3.in",
                        onComplete: () => {
                            menu.style.visibility = "hidden";
                        }
                    });
                }
            });
        });
    });
</script>
